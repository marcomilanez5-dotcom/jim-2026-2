  <script>
    let jogosData = [];
    let diaSelecionado = "Todos";
    let tempoRestante = 30;

    function carregarJogos() {
      if (typeof google !== 'undefined' && google.script) {
        google.script.run.withSuccessHandler(data => {
          jogosData = data;
          renderAoVivo();
          renderJogos();
          tempoRestante = 30;
        }).getJogosData();
      }
    }

    // 1. Renderiza bloco "Ao Vivo"
    function renderAoVivo() {
      const container = document.getElementById("cardsAoVivo");
      const secao = document.getElementById("secaoAoVivo");
      container.innerHTML = "";

      const emAndamento = jogosData.filter(j => j.status === "Em Andamento" || j.status === "Ao Vivo");

      if (emAndamento.length > 0) {
        secao.classList.remove("hidden");
        emAndamento.forEach(j => {
          container.appendChild(criarCardJogo(j, true));
        });
      } else {
        secao.classList.add("hidden");
      }
    }

    // 2. Renderiza a lista principal com filtros
    function renderJogos() {
      const container = document.getElementById("listaJogos");
      const mod = document.getElementById("filterModalidade").value;
      const turma = document.getElementById("filterTurma").value;
      container.innerHTML = "";

      const jogosFiltrados = jogosData.filter(j => {
        const matchMod = mod === "Todas" || j.modalidade === mod;
        
        const t1Str = String(j.t1 || "").trim().toLowerCase();
        const t2Str = String(j.t2 || "").trim().toLowerCase();
        const turmaBusca = String(turma).trim().toLowerCase();
        
        const matchTurma = turma === "Todas" || t1Str === turmaBusca || t2Str === turmaBusca || t1Str.includes(turmaBusca) || t2Str.includes(turmaBusca);
        const matchDia = diaSelecionado === "Todos" || (j.horario && String(j.horario).toLowerCase().includes(diaSelecionado.toLowerCase()));
        
        return matchMod && matchTurma && matchDia;
      });

      if (jogosFiltrados.length === 0) {
        container.innerHTML = `<div class="text-center text-gray-500 py-10 bg-white rounded-xl shadow-sm border">Nenhum jogo encontrado para estes filtros.</div>`;
        return;
      }

      jogosFiltrados.forEach(j => {
        container.appendChild(criarCardJogo(j, false));
      });

      renderChaveamento(mod);
    }

    // 5. Função para criar o Card do Jogo com Troféu e Destaque Visual
    function criarCardJogo(j, isLive) {
      const enc = j.status === 'Encerrado';
      const win1 = enc && j.p1 > j.p2;
      const win2 = enc && j.p2 > j.p1;

      const card = document.createElement("div");
      card.className = `bg-white p-4 rounded-xl shadow-sm border ${isLive ? 'border-red-400 bg-red-50/20' : 'border-gray-200'}`;

      card.innerHTML = `
        <div class="flex justify-between items-center mb-2">
          <span class="text-xs font-bold text-blue-700 bg-blue-50 px-2.5 py-1 rounded-md">${j.modalidade}</span>
          <span class="text-xs font-semibold ${enc ? 'text-green-600 bg-green-50 px-2 py-0.5 rounded' : 'text-amber-600 bg-amber-50 px-2 py-0.5 rounded'}">
            ${enc ? 'Encerrado' : (j.horario || 'Em breve')}
          </span>
        </div>

        <div class="flex items-center justify-between bg-gray-50 p-3 rounded-lg mt-2">
          <!-- Time 1 -->
          <div class="w-5/12 text-center font-bold text-sm ${win1 ? 'text-blue-900 font-extrabold' : 'text-gray-800'}">
            ${win1 ? '<i class="fas fa-trophy text-yellow-500 mr-1"></i>' : ''} ${j.t1}
          </div>
          
          <!-- Placar -->
          <div class="w-2/12 flex items-center justify-center gap-1.5 font-extrabold text-lg">
            <span class="${win1 ? 'text-blue-600 text-xl' : 'text-gray-700'}">${j.p1}</span>
            <span class="text-gray-400 font-normal">x</span>
            <span class="${win2 ? 'text-blue-600 text-xl' : 'text-gray-700'}">${j.p2}</span>
          </div>

          <!-- Time 2 -->
          <div class="w-5/12 text-center font-bold text-sm ${win2 ? 'text-blue-900 font-extrabold' : 'text-gray-800'}">
            ${j.t2} ${win2 ? '<i class="fas fa-trophy text-yellow-500 ml-1"></i>' : ''}
          </div>
        </div>
      `;
      return card;
    }

    // 4. Renderiza Árvore Visual Simples do Mata-Mata quando selecionada uma modalidade
    function renderChaveamento(mod) {
      const container = document.getElementById("bracketContainer");
      const secao = document.getElementById("secaoChaveamento");

      if (mod === "Todas") {
        secao.classList.add("hidden");
        return;
      }

      const jogosMod = jogosData.filter(j => j.modalidade === mod);
      const semis = jogosMod.filter(j => (j.fase || '').toLowerCase().includes("semi"));
      const final = jogosMod.filter(j => (j.fase || '').toLowerCase().includes("final"));

      if (semis.length === 0 && final.length === 0) {
        secao.classList.add("hidden");
        return;
      }

      secao.classList.remove("hidden");
      container.innerHTML = "";

      let htmlSemis = `<div class="space-y-2 w-full md:w-1/2"><h3 class="text-xs font-bold text-gray-400 uppercase text-center">Semifinais</h3>`;
      semis.forEach(s => {
        htmlSemis += `
          <div class="bg-gray-50 border p-2 rounded-lg text-xs flex justify-between font-bold">
            <span>${s.t1} vs ${s.t2}</span>
            <span class="text-blue-600">${s.p1} x ${s.p2}</span>
          </div>`;
      });
      htmlSemis += `</div>`;

      let htmlFinal = `<div class="w-full md:w-1/2"><h3 class="text-xs font-bold text-yellow-600 uppercase text-center mb-2"><i class="fas fa-crown"></i> Grande Final</h3>`;
      final.forEach(f => {
        htmlFinal += `
          <div class="bg-yellow-50 border border-yellow-300 p-3 rounded-lg text-xs flex justify-between font-extrabold text-gray-900">
            <span>${f.t1} vs ${f.t2}</span>
            <span class="text-blue-600 text-sm">${f.p1} x ${f.p2}</span>
          </div>`;
      });
      htmlFinal += `</div>`;

      container.innerHTML = htmlSemis + htmlFinal;
    }

    function setDia(dia) {
      diaSelecionado = dia;
      document.querySelectorAll(".btn-dia").forEach(btn => {
        if(btn.innerText.includes(dia)) {
          btn.className = "btn-dia px-3 py-1.5 text-xs font-bold rounded-lg bg-blue-600 text-white";
        } else {
          btn.className = "btn-dia px-3 py-1.5 text-xs font-bold rounded-lg bg-gray-100 text-gray-700 hover:bg-gray-200";
        }
      });
      renderJogos();
    }

    // Cronómetro de atualização (Melhoria 5)
    setInterval(() => {
      tempoRestante--;
      if (tempoRestante <= 0) {
        carregarJogos();
      } else {
        const timerEl = document.getElementById("timer");
        if(timerEl) timerEl.innerText = tempoRestante;
      }
    }, 1000);

    carregarJogos();
  </script>
